// Compact Smart Contract Definition
pragma language_version 0.23;
import CompactStandardLibrary;

export ledger visitorCount: Counter;
export ledger verifierId: Bytes<32>;
export ledger lastVisitorCommitment: Bytes<32>;

witness secretPasscode(): Bytes<32>;

export circuit verifyVisitor(expectedVerifier: Bytes<32>): Bytes<32> {
  assert(verifierId == expectedVerifier, "Invalid verifier ID provided");
  const passcode = secretPasscode();
  const visitorCommitment = persistentHash<Vector<2, Bytes<32>>>([
    pad(32, "vvp:visitor:commitment"),
    passcode
  ]);
  visitorCount.increment(1);
  const disclosedCommitment = disclose(visitorCommitment);
  lastVisitorCommitment = disclosedCommitment;
  return disclosedCommitment;
}
